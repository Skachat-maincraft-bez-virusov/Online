#Чтобы запустить библиотеку, введите в командную строку MSYS2 MINGW64 следующее:
cd c:\biblioteki
gcc -c list.c -o list.o #компиляция исходного файла list.c
gcc -c trees.c -o trees.o #компиляция исходного файла trees.c
ar rcs libmylib.a list.o trees.o #создание архива библиотеки
gcc main.c -L. -lmylib -o program #указание используемых в программе статических библиотек
./program #запуск
